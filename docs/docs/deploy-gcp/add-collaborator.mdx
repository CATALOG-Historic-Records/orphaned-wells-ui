import React, {useEffect, useState} from 'react';
import CodeBlock from '@theme/CodeBlock';

export const COLLABORATOR_EVENT = 'ogrre:add-collaborator-key-change';
export const COLLABORATOR_STORAGE_KEY = 'ogrre:add-collaborator-key';
export const COLLABORATOR_TOKEN = '__COLLABORATOR__';
export const COLLABORATOR_UPPER_TOKEN = '__COLLABORATOR_UPPER__';

export function normalizeCollaborator(value) {
  return value.trim().toLowerCase();
}

export function getCollaboratorValues(value) {
  const collaborator = normalizeCollaborator(value);

  return {
    collaborator,
    collaboratorValue: collaborator || '<collaborator>',
    collaboratorUpperValue: collaborator ? collaborator.toUpperCase() : '<COLLABORATOR>',
  };
}

export function useCollaboratorValues() {
  const [rawCollaborator, setRawCollaborator] = useState('');

  useEffect(() => {
    const storedCollaborator = window.sessionStorage.getItem(COLLABORATOR_STORAGE_KEY) || '';

    setRawCollaborator(storedCollaborator);

    function handleCollaboratorChange(event) {
      setRawCollaborator(event.detail || '');
    }

    window.addEventListener(COLLABORATOR_EVENT, handleCollaboratorChange);

    return () => {
      window.removeEventListener(COLLABORATOR_EVENT, handleCollaboratorChange);
    };
  }, []);

  return getCollaboratorValues(rawCollaborator);
}

export function CollaboratorKeyInput() {
  const [rawCollaborator, setRawCollaborator] = useState('');

  useEffect(() => {
    setRawCollaborator(window.sessionStorage.getItem(COLLABORATOR_STORAGE_KEY) || '');
  }, []);

  function handleChange(event) {
    const nextValue = event.target.value;

    setRawCollaborator(nextValue);
    window.sessionStorage.setItem(COLLABORATOR_STORAGE_KEY, nextValue);
    window.dispatchEvent(new CustomEvent(COLLABORATOR_EVENT, {detail: nextValue}));
  }

  return (
    <input
      aria-label="Collaborator key"
      onChange={handleChange}
      placeholder="<collaborator>"
      type="text"
      value={rawCollaborator}
    />
  );
}

export function CollaboratorInline({prefix = '', suffix = '', uppercase = false}) {
  const {collaboratorValue, collaboratorUpperValue} = useCollaboratorValues();
  const value = uppercase ? collaboratorUpperValue : collaboratorValue;

  return <code>{prefix}{value}{suffix}</code>;
}

export function renderCollaboratorTemplate(template, values) {
  return template
    .replaceAll(COLLABORATOR_UPPER_TOKEN, values.collaboratorUpperValue)
    .replaceAll(COLLABORATOR_TOKEN, values.collaboratorValue);
}

export function CollaboratorCodeBlock({children, language}) {
  const values = useCollaboratorValues();
  const template = React.Children.toArray(children).join('').trim();

  return (
    <CodeBlock language={language}>
      {renderCollaboratorTemplate(template, values)}
    </CodeBlock>
  );
}

export function KubernetesTargetLookupCodeBlock() {
  const {collaborator, collaboratorValue} = useCollaboratorValues();
  const command = collaborator
    ? `terraform workspace select ogrre
terraform output -json kubernetes_deploy_targets | jq '.[\"${collaborator}\"]'`
    : `terraform workspace select ogrre
terraform output -json kubernetes_deploy_targets | jq --arg collaborator "${collaboratorValue}" '.[$collaborator]'`;

  return <CodeBlock language="bash">{command}</CodeBlock>;
}

# Add Collaborator Deployment

Use this checklist when adding a new public OGRRE collaborator deployment using the standard GitHub Actions workflow and default settings.

Enter the short lowercase collaborator key below (such as `isgs`, `newts`, `osage`, or `rrc`) to populate all commands and resource names on this page.

## 0. Prerequisites

- Local CLI tools: `gh` and `gcloud`.
- Access to the target GCP project and GitHub repository settings (`CATALOG-Historic-Records`).
- See [Backend on GKE — Prerequisites](./backend-gke#prerequisites) for full GCP API and service-account permission details.

## 2. Configure and Deploy the Backend (`orphaned-wells-ui-server`)

1. **Add the collaborator to `terraform.tfvars` and sync to CI**:
   Pull the shared `terraform.tfvars` configuration from GCS:
   ```bash
   ./deployment/terraform/scripts/sync_tfvars.sh pull
   ```
   Add <CollaboratorInline /> under `gke_backends` in `deployment/terraform/terraform.tfvars`:
   <CollaboratorCodeBlock language="hcl">{`
   gke_backends = {
     # existing collaborators...
     "__COLLABORATOR__" = {}
   }
   `}</CollaboratorCodeBlock>
   Upload the updated file to GCS and sync the `PROD_TERRAFORM_TFVARS` GitHub secret:
   ```bash
   ./deployment/terraform/scripts/sync_tfvars.sh push --secret
   ```

2. **Register the collaborator in `collaborators.json`**:
   Add <CollaboratorInline /> to `collaborators.json` at the root of `orphaned-wells-ui-server`:
   <CollaboratorCodeBlock language="json">{`
   [
     "osage",
     "newts",
     "isgs",
     "rrc",
     "__COLLABORATOR__"
   ]
   `}</CollaboratorCodeBlock>

3. **Merge to `main` and approve Terraform in GitHub Actions**:
   Commit the `collaborators.json` update and merge it to `main`. Merging to `main` triggers the **Deploy Staging Server to GKE** workflow, which generates a Terraform plan for the new collaborator's bucket, namespace, static IP, and DNS records:
   1. In GitHub Actions (`CATALOG-Historic-Records/orphaned-wells-ui-server`), open the **Deploy Staging Server to GKE** run.
   2. Review the **Terraform plan** summary on the workflow run page.
   3. **If everything looks correct**, click **Review deployments → `terraform-apply` → Approve and deploy** and wait for the infrastructure apply job to finish before proceeding. *(Note: The `terraform-apply` GitHub Environment is restricted to required reviewers; if you do not have permission to approve it, ask a repository owner to review and approve the plan.)*

4. **Create and push the collaborator branch**:
   Create the <CollaboratorInline /> branch from `main` and push it to trigger `.github/workflows/deploy-k8s-collaborator.yml`:
   <CollaboratorCodeBlock language="bash">{`
   git checkout main && git pull
   git checkout -b __COLLABORATOR__ && git push -u origin __COLLABORATOR__
   `}</CollaboratorCodeBlock>
   The workflow automatically provisions the <CollaboratorInline prefix="ogrre-" /> MongoDB database and user (on first deploy), injects all shared runtime/OAuth/service-account credentials, and deploys the backend to GKE (see [What the backend workflow configures automatically](#what-the-backend-workflow-configures-automatically) below).

   Once the workflow finishes and the Google-managed TLS certificate activates, verify the health endpoint:
   <CollaboratorCodeBlock language="bash">{`
   curl -f https://__COLLABORATOR__-server.uow-carbon.org/health
   `}</CollaboratorCodeBlock>

## 3. Configure and Deploy the Frontend (`orphaned-wells-ui`)

1. **Create the GitHub Environment and `BACKEND_URL` secret**:
   Configure `BACKEND_URL` (with no trailing slash) in the <CollaboratorInline /> GitHub Environment:
   <CollaboratorCodeBlock language="bash">{`
   gh api --method PUT repos/CATALOG-Historic-Records/orphaned-wells-ui/environments/__COLLABORATOR__

   gh secret set BACKEND_URL \\
     -R CATALOG-Historic-Records/orphaned-wells-ui \\
     --env __COLLABORATOR__ \\
     --body "https://__COLLABORATOR__-server.uow-carbon.org"
   `}</CollaboratorCodeBlock>

2. **Update `collaborators.json` and `dispatch.yaml`**:
   Add <CollaboratorInline /> to `collaborators.json` at the root of `orphaned-wells-ui`:
   <CollaboratorCodeBlock language="json">{`
   [
     "isgs",
     "newts",
     "osage",
     "rrc",
     "__COLLABORATOR__"
   ]
   `}</CollaboratorCodeBlock>
   Add the route for <CollaboratorInline suffix=".uow-carbon.org/*" /> to `deployment/app-engine/dispatch.yaml`:
   <CollaboratorCodeBlock language="yaml">{`
   - url: "__COLLABORATOR__.uow-carbon.org/*"
     service: __COLLABORATOR__-uow
   `}</CollaboratorCodeBlock>
   Merge both changes to `main`.

3. **Create and push the collaborator branch**:
   Create the <CollaboratorInline /> branch from `main` and push it to trigger `.github/workflows/deploy-collaborator.yml`:
   <CollaboratorCodeBlock language="bash">{`
   git checkout main && git pull
   git checkout -b __COLLABORATOR__ && git push -u origin __COLLABORATOR__
   `}</CollaboratorCodeBlock>
   The workflow automatically renders `service: `<CollaboratorInline suffix="-uow" /> using `deployment/app-engine/app.yaml`, builds the React bundle, and deploys the App Engine service.

## 4. Complete GCP Routing, DNS, and OAuth

After the frontend App Engine service is deployed, complete these one-time GCP configuration steps:

1. **Deploy `dispatch.yaml`**:
   ```bash
   cd orphaned-wells-ui/deployment/app-engine
   gcloud app deploy dispatch.yaml
   ```
2. **Add Frontend Cloud DNS Records**:
   In Google Cloud DNS, add `A` (IPv4) and `AAAA` (IPv6) records for <CollaboratorInline suffix=".uow-carbon.org" /> using the same App Engine addresses as the existing frontend instances.
3. **Add the App Engine Custom Domain**:
   In the Google Cloud Console, open **App Engine → Settings → Custom domains** (or **Domain mappings**) and map <CollaboratorInline suffix=".uow-carbon.org" /> to the <CollaboratorInline suffix="-uow" /> service.
4. **Update Google OAuth Credentials**:
   In **APIs & Services → Credentials**, add <CollaboratorInline prefix="https://" suffix=".uow-carbon.org" /> to the OAuth client's **Authorized JavaScript Origins** and **Authorized Redirect URIs**.

## Optional Customizations and Manual Fallbacks

### What the backend workflow configures automatically

When `.github/workflows/deploy-k8s-collaborator.yml` (via `deploy-k8s-dispatch.yml`) runs for <CollaboratorInline />, it automatically handles:
- **First-time MongoDB provisioning**: On the first deployment (when no `DB_*` credentials exist in `RUNTIME_ENV`, GCS, or GKE), the workflow uses the repository's MongoDB Atlas API secrets to create a scoped database user (<CollaboratorInline prefix="ogrre-" />) with `readWrite` access only on database <CollaboratorInline prefix="ogrre-" />, verifies the database has zero existing collections, and seeds `roles`, initial team <CollaboratorInline uppercase />, and admin users (`DEFAULT_ADMIN_EMAILS`). On subsequent deployments, it reuses those existing database credentials automatically.
- **Service account `.json` files & Google OAuth**: Mounts `STORAGE_SERVICE_KEY_JSON` and `DOCUMENT_AI_SERVICE_KEY_JSON` into `/code/ogrre/` and populates `client_id` / `client_secret` from `GOOGLE_CLIENT_ID` / `GOOGLE_CLIENT_SECRET` (no `.json` key files or backend `RUNTIME_ENV` secret need to be uploaded unless overriding defaults).
- **Host, CORS, session cookies, and `.env` snapshot**: Sets `ALLOWED_ORIGINS`, `SESSION_COOKIE_*`, `STORAGE_BUCKET_NAME`, `USE_DB_SCHEMAS=false`, and worker sizing from the Terraform target, applies `Secret/backend-runtime-env` to GKE, and saves the complete active `.env` file to <CollaboratorInline prefix="gs://tidy-outlet-412020-ogrre-terraform-state/secrets/" suffix=".env" />.

### Inspecting active runtime `.env` snapshots in GCS

Every backend deployment saves the complete active `.env` configuration rendered for that instance to Google Cloud Storage. To inspect or download the active `.env` for <CollaboratorInline />:

<CollaboratorCodeBlock language="bash">{`
gcloud storage cat gs://tidy-outlet-412020-ogrre-terraform-state/secrets/__COLLABORATOR__.env
`}</CollaboratorCodeBlock>

### Custom `RUNTIME_ENV` overrides or manual MongoDB setup

If you need to override default backend runtime variables (for example, setting `USE_DB_SCHEMAS=true`, configuring CILogon credentials, or supplying custom MongoDB credentials when not using automatic MongoDB Atlas provisioning), configure a `RUNTIME_ENV` secret in the <CollaboratorInline /> GitHub Environment before deploying the branch:

1. **(Only when not using automatic Atlas provisioning)** Create a MongoDB database and user, then download <a href="/orphaned-wells-ui/downloads/InitializeMongoDB.py" download>InitializeMongoDB.py</a> (see [Create and connect MongoDB database](/docs/install/database_setup#initialize-database-for-ogrre)) and run it once to seed `roles`, `teams`, and `users` (the script refuses to overwrite any database that already contains collections unless `--force` is passed):
   ```bash
   python InitializeMongoDB.py \
     --team_name "<team name>" \
     --email "<initial-admin-email>" \
     --database_username "<DB_USERNAME>" \
     --database_password "<DB_PASSWORD>" \
     --database_hostname "<DB_CONNECTION>" \
     --database_name "<DB_NAME>"
   ```
   *(For self-hosted or non-Atlas MongoDB deployments, `--database_hostname` and `DB_CONNECTION` also accept a full `mongodb://` or `mongodb+srv://` URI without separate `--database_username` / `--database_password` flags.)*

2. Create a local <CollaboratorInline suffix=".env" /> file with your overrides or manual database credentials and upload it to the <CollaboratorInline /> GitHub Environment:
   <CollaboratorCodeBlock language="bash">{`
   gh api --method PUT repos/CATALOG-Historic-Records/orphaned-wells-ui-server/environments/__COLLABORATOR__

   gh secret set RUNTIME_ENV \\
     -R CATALOG-Historic-Records/orphaned-wells-ui-server \\
     --env __COLLABORATOR__ < path/to/__COLLABORATOR__.env
   `}</CollaboratorCodeBlock>
   Explicit values in `RUNTIME_ENV` always take precedence over the workflow's defaults and over any previously saved GCS/GKE snapshot.

### Custom Terraform resource sizing or bucket names

By default, a collaborator backend uses `<collaborator>_uploads` for its GCS bucket, 2 API replicas at 1 CPU / 4 GiB memory per pod, and 1850m CPU / 12 GiB memory for batch-processing workers.

To customize these settings (for example, to reuse an existing bucket name or reduce sizing for a smaller test instance), specify optional attributes in `gke_backends` inside `terraform.tfvars`:

<CollaboratorCodeBlock language="hcl">{`
gke_backends = {
  # existing collaborators...
  "__COLLABORATOR__" = {
    upload_bucket_name   = "existing-bucket-name"
    replicas             = 2
    cpu_request          = "1"
    memory_request       = "4Gi"
    cpu_limit            = "1"
    memory_limit         = "4Gi"
    persistent_disk_size = "20Gi"
  }
}
`}</CollaboratorCodeBlock>

For GKE deployments, the workflow overwrites `ENVIRONMENT`, `BACKEND_URL`, and `STORAGE_BUCKET_NAME` from the Terraform deployment target; always configure custom bucket names in `terraform.tfvars` rather than `RUNTIME_ENV`.

### GCS bucket IAM and external batch buckets

Because `ogrre-storage-runtime` and `ogrre-document-ai` have project-level `roles/storage.objectUser` (and `ogrre-document-ai` has project-level Document AI roles), newly created <CollaboratorInline suffix="_uploads" /> buckets and any arbitrary buckets in the same GCP project inherit access automatically. You only need to grant `roles/storage.objectUser` manually if a collaborator runs GCS batch uploads from an external bucket in a different GCP project.

### Manual dispatch dropdown in `deploy-k8s-dispatch.yml`

`.github/workflows/deploy-k8s-collaborator.yml` supports `workflow_dispatch` on any collaborator branch listed in `collaborators.json` without editing workflow files. If you also want <CollaboratorInline /> to appear in the `DEPLOY_ENV` choice dropdown when manually triggering `.github/workflows/deploy-k8s-dispatch.yml` from the GitHub Actions UI, add <CollaboratorInline /> to `workflow_dispatch.inputs.DEPLOY_ENV.options` in `.github/workflows/deploy-k8s-dispatch.yml`.

### Inspecting Kubernetes resources with `kubectl`

To run `kubectl` commands against the GKE cluster (see [Backend on GKE — Check rollout status](./backend-gke#check-rollout-status)), install the GKE auth plugin and fetch cluster credentials once using a Google account with GKE access in GCP IAM (such as `OGRRE_Dev`, `roles/container.developer`, or `roles/container.admin`):

<CollaboratorCodeBlock language="bash">{`
gcloud auth login
gcloud components install gke-gcloud-auth-plugin
gcloud container clusters get-credentials uow-backend-gke \
  --region us-central1 \
  --project tidy-outlet-412020

kubectl -n uow-__COLLABORATOR__ get deployment backend
kubectl -n uow-__COLLABORATOR__ get pods -l app.kubernetes.io/name=orphaned-wells-ui-server -o wide
kubectl -n uow-__COLLABORATOR__ get managedcertificate backend-cert
`}</CollaboratorCodeBlock>

### Manual Terraform fallback (only when `ENABLE_TERRAFORM_CI` is disabled)

With `ENABLE_TERRAFORM_CI=true`, Terraform planning, applying, and reading `kubernetes_deploy_targets` happen automatically in GitHub Actions. Only use the local commands below when Terraform CI automation is disabled:

```bash
cd orphaned-wells-ui-server/deployment/terraform
terraform init
terraform workspace select ogrre

  --repo CATALOG-Historic-Records/orphaned-wells-ui-server \

```

## Common pitfalls

- **DNS vs. Ingress host**: Updating DNS alone is not enough for GKE; the Kubernetes Ingress `host` and `ManagedCertificate` domain are rendered from the Terraform target and require a backend deployment after Terraform apply.
- **No trailing slashes**: Frontend `BACKEND_URL` secrets and OAuth origin entries should not have trailing slashes.
- **Keep both `collaborators.json` files updated**: Ensure <CollaboratorInline /> is added to `collaborators.json` in both `orphaned-wells-ui-server` and `orphaned-wells-ui` so push deployments and `main → `<CollaboratorInline /> sync PRs are active across both repositories.
